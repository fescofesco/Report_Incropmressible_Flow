\section{Problem Description}

\subsection{Geometry and Setup}

The problem considers two-dimensional laminar flow in a cylindrical pipe with diameter~$D$ and length $L = 50D$.
The pipe wall is heated with constant heat flux $q_w = \text{const}$.
The incompressible fluid has constant properties: density~$\rho$, dynamic viscosity~$\mu$, thermal conductivity~$\lambda$, and specific heat capacity~$c_v$.
Viscous dissipation is neglected.

The flow enters with uniform velocity and temperature profiles:
\begin{align}
    w\big|_{z=0} &= W_{\text{in}} = \text{const.} \\
    T\big|_{z=0} &= T_{\text{in}} = \text{const.}
\end{align}

\begin{figure}[H]
    \centering
    \includegraphics[width=0.95\textwidth]{../Heun/Plots_python/geometry_diagram.png}
    \caption{Schematic of the cylindrical pipe flow with constant wall heat flux.
             Flow enters with uniform velocity $W_{\text{in}}$ and temperature $T_{\text{in}}$,
             and develops into a parabolic profile along the pipe length.}
    \label{fig:geometry}
\end{figure}

\subsection{Physical Parameters}

\begin{itemize}
    \item Reynolds number: $\Rey = \dfrac{\rho W_{\text{in}} D}{\mu} = 100$
    \item Prandtl number: $\Pran = \dfrac{\mu c_v}{\lambda} = 5$
    \item Length-to-diameter ratio: $L/D = 50$
\end{itemize}

% 2. MATHEMATICAL FORMULATION

\section{Mathematical Formulation}

\subsection{Governing Equations in Cylindrical Coordinates}

For axisymmetric flow in cylindrical coordinates $(r, z)$ with velocity components $(u, w)$ in the radial and axial directions, the governing equations in \textbf{conservation form} are:

\subsubsection{Continuity Equation}

\begin{equation}
    \underbrace{\frac{1}{r}\parder{(r\,u)}{r}}_{\text{radial mass-flux divergence}}
    + \underbrace{\parder{w}{z}}_{\text{axial mass-flux divergence}} = 0
    \label{eq:continuity_dim}
\end{equation}
The $1/r$ weighting on the radial term accounts for the circumferential area
$2\pi r$ growing with radius; the two terms summing to zero enforces
incompressibility (no net accumulation of mass in any control volume).

\subsubsection{Momentum Equations}

\textbf{Radial momentum:}
\begin{equation}
    \parder{u}{t} + \frac{1}{r}\parder{(r\,u^2)}{r} + \parder{(u\,w)}{z}
    = -\frac{1}{\rho}\parder{p}{r}
      + \frac{1}{\rho}\left[\frac{1}{r}\parder{(r\,\tau_{rr})}{r} - \frac{\tau_{\varphi\varphi}}{r} + \parder{\tau_{rz}}{z}\right]
    \label{eq:momentum_r_dim}
\end{equation}

\textbf{Axial momentum:}
\begin{equation}
    \parder{w}{t} + \frac{1}{r}\parder{(r\,u\,w)}{r} + \parder{(w^2)}{z}
    = -\frac{1}{\rho}\parder{p}{z}
      + \frac{1}{\rho}\left[\frac{1}{r}\parder{(r\,\tau_{rz})}{r} + \parder{\tau_{zz}}{z}\right]
    \label{eq:momentum_z_dim}
\end{equation}

Term by term, each momentum equation reads (local rate of change) + (convection)
= (pressure force) + (viscous force):
\begin{itemize}
    \item $\parder{u}{t}$, $\parder{w}{t}$ --- \textbf{local (unsteady) acceleration}: the rate of
          change of momentum stored in a fixed control volume.
    \item $\frac{1}{r}\parder{(r\,u^2)}{r}$, $\parder{(u\,w)}{z}$ (and the analogous axial terms)
          --- \textbf{convective transport}: momentum carried in and out of the control volume by
          the flow itself (advection by $u$ and $w$).
    \item $-\frac{1}{\rho}\parder{p}{r}$, $-\frac{1}{\rho}\parder{p}{z}$ --- \textbf{pressure-gradient
          force} per unit mass, the driving force pushing fluid from high to low pressure.
    \item $\frac{1}{\rho}\big[\frac1r\parder{(r\tau_{rr})}{r}+\parder{\tau_{rz}}{z}\big]$ and the
          axial counterpart --- \textbf{viscous (diffusive) force} per unit mass, the net effect
          of internal friction spreading momentum from fast- to slow-moving fluid.
    \item $-\dfrac{\tau_{\varphi\varphi}}{\rho r}$ --- a \textbf{curvature (``hoop-stress'') term}
          unique to cylindrical coordinates: it arises because the circumferential stress
          $\tau_{\varphi\varphi}$ acts on a surface whose area changes with $r$; there is no
          Cartesian analogue.
\end{itemize}

The stress tensor components for a Newtonian fluid are:
\begin{equation}
    \underbrace{\tau_{rr} = 2\mu\parder{u}{r}}_{\text{radial normal stress}}, \quad
    \underbrace{\tau_{\varphi\varphi} = 2\mu\frac{u}{r}}_{\text{hoop (circumferential) normal stress}}, \quad
    \underbrace{\tau_{zz} = 2\mu\parder{w}{z}}_{\text{axial normal stress}}, \quad
    \underbrace{\tau_{rz} = \mu\left(\parder{u}{z} + \parder{w}{r}\right)}_{\text{shear stress}}
\end{equation}

\subsubsection{Energy Equation}

\begin{equation}
    \underbrace{\parder{T}{t}}_{\text{local rate of change}}
    + \underbrace{\frac{1}{r}\parder{(r\,u\,T)}{r} + \parder{(w\,T)}{z}}_{\text{convective transport of heat by the flow}}
    = \underbrace{\frac{\lambda}{\rho c_v}\left[\frac{1}{r}\parder{}{r}\!\left(r\parder{T}{r}\right) + \pparder{T}{z}\right]}_{\text{conductive (diffusive) transport, Fourier's law}}
    \label{eq:energy_dim}
\end{equation}
Viscous dissipation (heat generated by internal friction) is neglected, per the problem statement.

\subsection{Boundary Conditions}

\textbf{Inlet} ($z = 0$):
\begin{equation}
    u = 0, \quad w = W_{\text{in}}, \quad T = T_{\text{in}}
\end{equation}

\textbf{Wall} ($r = D/2$):
\begin{equation}
    u = 0, \quad w = 0, \quad \lambda\parder{T}{r}\bigg|_{r=D/2} = q_w
\end{equation}
Heat flows into the fluid at the wall (constant heating), so with $r$ increasing outward this is a positive gradient: the wall is hotter than the centreline.

\textbf{Centreline} ($r = 0$, symmetry):
\begin{equation}
    u = 0, \quad \parder{w}{r}\bigg|_{r=0} = 0, \quad \parder{T}{r}\bigg|_{r=0} = 0
\end{equation}

\textbf{Outlet} ($z = L$, zero-gradient):
\begin{equation}
    \parder{u}{z} = 0, \quad \parder{w}{z} = 0, \quad \parder{T}{z} = 0
\end{equation}

\newpage

\subsection{Non-Dimensionalization}

\subsubsection{Reference Scales}
\begin{itemize}
    \item Length: $D$
    \item Velocity: $W_{\text{in}}$
    \item Pressure: $\rho W_{\text{in}}^2$
    \item Temperature difference: $\dfrac{q_w}{\rho c_v W_{\text{in}}}$
    \item Time: $D / W_{\text{in}}$
\end{itemize}

\subsubsection{Non-Dimensional Variables}
\begin{align}
    r^* &= \frac{r}{D}, \quad z^* = \frac{z}{D}, \quad t^* = \frac{t\,W_{\text{in}}}{D} \\[6pt]
    u^* &= \frac{u}{W_{\text{in}}}, \quad w^* = \frac{w}{W_{\text{in}}}, \quad
    p^* = \frac{p}{\rho W_{\text{in}}^2} \\[6pt]
    \theta &= \frac{(T - T_{\text{in}})\,\rho\, c_v\, W_{\text{in}}}{q_w}
\end{align}

\subsubsection{Non-Dimensional Governing Equations}

Dropping asterisks for brevity (all quantities are non-dimensional hereafter):

\textbf{Continuity:}
\begin{equation}
    \frac{1}{r}\parder{(r\,u)}{r} + \parder{w}{z} = 0
    \label{eq:continuity_nd}
\end{equation}

\textbf{Radial momentum:}
\begin{equation}
    \parder{u}{t} + \frac{1}{r}\parder{(r\,u^2)}{r} + \parder{(u\,w)}{z}
    = -\parder{p}{r}
      + \frac{1}{\Rey}\left[\frac{1}{r}\parder{}{r}\!\left(r\parder{u}{r}\right) - \frac{u}{r^2} + \pparder{u}{z}\right]
    \label{eq:momentum_r_nd}
\end{equation}

\textbf{Axial momentum:}
\begin{equation}
    \parder{w}{t} + \frac{1}{r}\parder{(r\,u\,w)}{r} + \parder{(w^2)}{z}
    = -\parder{p}{z}
      + \frac{1}{\Rey}\left[\frac{1}{r}\parder{}{r}\!\left(r\parder{w}{r}\right) + \pparder{w}{z}\right]
    \label{eq:momentum_z_nd}
\end{equation}

\textbf{Energy:}
\begin{equation}
    \parder{\theta}{t} + \frac{1}{r}\parder{(r\,u\,\theta)}{r} + \parder{(w\,\theta)}{z}
    = \frac{1}{\Rey\Pran}\left[\frac{1}{r}\parder{}{r}\!\left(r\parder{\theta}{r}\right) + \pparder{\theta}{z}\right]
    \label{eq:energy_nd}
\end{equation}

\subsubsection{Non-Dimensional Boundary Conditions}

\textbf{Inlet} ($z = 0$):
$\;u = 0,\; w = 1,\; \theta = 0$

\textbf{Wall} ($r = 1/2$):
$\;u = 0,\; w = 0,\; \partial\theta/\partial r = \Rey\Pran$

Non-dimensionalizing the wall condition directly: $\partial\theta/\partial r^* = D\cdot(\rho c_v W_{\text{in}}/q_w)\cdot\partial T/\partial r|_{\text{wall}} = D\cdot(\rho c_v W_{\text{in}}/q_w)\cdot(q_w/\lambda) = \rho c_v W_{\text{in}} D/\lambda = \Rey\Pran$ (i.e.\ 500 for $\Rey=100,\Pran=5$), \emph{not} $\pm 1$ — the reference length/temperature scales do not cancel.

\textbf{Centreline} ($r = 0$):
$\;u = 0,\; \partial w/\partial r = 0,\; \partial\theta/\partial r = 0$

\textbf{Outlet} ($z = 50$):
$\;\partial u/\partial z = 0,\; \partial w/\partial z = 0,\; \partial\theta/\partial z = 0$

\begin{equation}
    \Omega = \{(r, z) : 0 \leq r \leq 0.5, \quad 0 \leq z \leq 50\}

\begin{itemize}

    \item Axial velocity $w$: axial faces $(r_{c,i},\, z_{f,j})$

\begin{equation}
    \Delta r = \frac{0.5}{n_r} = 0.01, \qquad \Delta z = \frac{50}{n_z} = 0.2

\subsection{Spatial Discretization: Finite Volume Method}

The governing equations are integrated over each control volume using the FVM.
After applying Gauss' divergence theorem, the volume integrals become surface flux integrals that are evaluated at control-volume faces.

\subsubsection{Continuity (at scalar CV)}

\begin{equation}
    \frac{r_{i+\frac{1}{2}}\,u_{i+1,j} - r_{i-\frac{1}{2}}\,u_{i,j}}{r_{c,i}\,\Delta r}
    + \frac{w_{i,j+1} - w_{i,j}}{\Delta z} = 0
    \label{eq:div_discrete}
\end{equation}

\subsubsection{Convective Scheme: 2nd-Order Upwind (Linear Upwind Differencing)}

\paragraph{What an upwind scheme is.} In the finite volume method the transported quantity
$\phi$ ($w$, $u$ or $\theta$) is stored at cell centres, but the convective flux
$F=v\,\phi$ has to be evaluated at the cell \emph{faces}, which lie between two centres. The
face value $\phi_{\text{face}}$ must therefore be interpolated from neighbouring cells, and the
choice of this interpolation is the convective scheme.

The obvious choice is \textbf{central differencing}, the average of the two neighbours,
\begin{equation}
    \phi_{\text{face}} = \tfrac12\left(\phi_i + \phi_{i+1}\right).
\end{equation}
It is second-order accurate but treats both sides of the face equally. Convection, however, is
directional: information is carried \emph{with} the flow, so the value arriving at a face comes
from the upstream side. When convection dominates diffusion, central differencing produces
non-physical oscillations (over- and undershoots). The governing parameter is the cell Peclet
number, the ratio of convective to diffusive transport across one cell,
\begin{equation}
    \text{Pe} = \frac{|v|\,\Delta x}{\nu_{\text{eff}}},
\end{equation}
and central differencing only remains bounded for $\text{Pe} < 2$. For our grid
($\Delta z = 0.2$, $\Delta r = 0.01$), based on the mean axial velocity $|w|=1$,
\begin{equation}
    \text{Pe}_z = |w|\,\Delta z\,\Rey = 1 \times 0.2 \times 100 = 20 \gg 2
\end{equation}
and $\text{Pe}_\theta = |w|\,\Delta z\,\Rey\Pran = 100$ for the energy equation. On the
centreline, where $w\approx2$, these values double to $40$ and $200$. Pure central differencing
is therefore unsuitable for the convective terms of this problem.

\textbf{Upwind schemes} instead take the face value from the side the flow comes from, as
decided by the sign of the transporting velocity at the face:
\begin{itemize}
    \item \textbf{1st-order upwind}: $\phi_{\text{face}}=\phi_i$ if $v_{\text{face}}>0$ (flow
          from cell $i$ towards $i+1$), otherwise $\phi_{\text{face}}=\phi_{i+1}$. It is very
          robust and does not oscillate, but it is only first-order accurate and introduces
          artificial (numerical) diffusion that smears gradients.
    \item \textbf{2nd-order upwind (linear upwind differencing, LUD)}, used in this work: the
          face value is linearly extrapolated from the \emph{two} upstream cells.
\end{itemize}
For a generic flux $F = v\,\phi$ at the face between cells $i$ and $i+1$, LUD reads
\begin{equation}
    \phi_{\text{face}} =
    \begin{cases}
        \tfrac{3}{2}\phi_i - \tfrac{1}{2}\phi_{i-1} & \text{if } v_{\text{face}} > 0
            \quad\text{(upstream = low-index side)} \\[4pt]
        \tfrac{3}{2}\phi_{i+1} - \tfrac{1}{2}\phi_{i+2} & \text{if } v_{\text{face}} < 0
            \quad\text{(upstream = high-index side)}
    \end{cases}
    \label{eq:lud}
\end{equation}
which reduces to the 1st-order upwind value ($\phi_i$ or $\phi_{i+1}$) if the linear
extrapolation term is dropped. Equation~\eqref{eq:lud} is implemented in
\texttt{upwind2\_face.m} (Listing~\ref{lst:upwind2}) and is called by all three right-hand-side
functions \texttt{compute\_rhs\_w.m}, \texttt{compute\_rhs\_u.m} and \texttt{compute\_rhs\_T.m}.
LUD keeps the upstream bias needed for stability at $\text{Pe}\gg2$, while on a uniform grid and
for a smooth solution its truncation error is $\mathcal{O}(\Delta x^2)$ (shown below), so the
convective discretization is formally second-order accurate, as required by the assignment.
Diffusive terms retain standard \textbf{second-order central differencing}, since diffusion has
no preferred direction.

\paragraph{Trade-off: LUD is not bounded.} Second-order accuracy is not a boundedness guarantee.
Because the extrapolated face value is not limited, LUD can overshoot or undershoot near sharp
transients, boundaries, or insufficiently resolved gradients; the scheme is second-order in
smooth regions but is not monotone or TVD everywhere. In the present simulation this appears
only as a small, localized, transient undershoot of $\theta$ near the inlet--wall corner during
the first time steps (Fig.~\ref{fig:T_profiles}); on coarser grids the undershoot would be
larger. Bounded alternatives are TVD schemes with flux limiters (e.g.\ minmod or van Leer),
which blend LUD with 1st-order upwind wherever the solution gradient is steep.

\paragraph{Why these reconstructions are second order.} Expand $\phi$ in a Taylor series about
the face location $x_f$ (face between cells $i$ and $i{+}1$, spacing $\Delta x$, upwind case
$v_{\text{face}}>0$): with $x_i = x_f-\tfrac12\Delta x$ and $x_{i-1}=x_f-\tfrac32\Delta x$,
\begin{equation}
    \phi_i = \phi_f - \tfrac12\Delta x\,\phi_f' + \tfrac18\Delta x^2\phi_f'' + \mathcal{O}(\Delta x^3),
    \qquad
    \phi_{i-1} = \phi_f - \tfrac32\Delta x\,\phi_f' + \tfrac98\Delta x^2\phi_f'' + \mathcal{O}(\Delta x^3).
\end{equation}
Plain 1st-order upwind ($\phi_{\text{face}}=\phi_i$) retains the $-\tfrac12\Delta x\,\phi_f'$ term
and is therefore only $\mathcal{O}(\Delta x)$ accurate. Forming the linear-upwind combination
$\tfrac32\phi_i-\tfrac12\phi_{i-1}$ cancels the $\phi_f'$ term \emph{exactly}
($-\tfrac32\cdot\tfrac12\Delta x+\tfrac12\cdot\tfrac32\Delta x=0$), leaving
$\phi_{\text{face}}=\phi_f-\tfrac38\Delta x^2\phi_f''+\mathcal{O}(\Delta x^3)$: the leading error is
$\mathcal{O}(\Delta x^2)$, confirming LUD is second order. The central-difference diffusion
operator is second order by the same mechanism applied symmetrically: expanding $\phi_{i\pm1}$
about $x_i$ and summing, the odd-order ($\Delta x$, $\Delta x^3$) terms cancel by symmetry,
leaving $(\phi_{i+1}-2\phi_i+\phi_{i-1})/\Delta x^2 = \phi_i''+\tfrac{1}{12}\Delta x^2\phi_i''''+\mathcal{O}(\Delta x^4)$.

\subsubsection{Temperature equation and cell-by-cell development}

The temperature variable $\theta_{i,j}$ is stored at the centre of every scalar control
volume. At each time step, its value is advanced from the net transport and diffusion through
the four faces of that control volume:
\begin{equation}
        \frac{\mathrm{d}\theta_{i,j}}{\mathrm{d}t}
        = -\frac{1}{r_{c,i}\Delta r}
            \left[(r u\theta)_e-(r u\theta)_w\right]
            -\frac{(w\theta)_n-(w\theta)_s}{\Delta z}
            +\frac{1}{\Rey\Pran}\left[
            \frac{r_{i+\frac12}\theta_r|_e-r_{i-\frac12}\theta_r|_w}{r_{c,i}\Delta r}
            +\frac{\theta_z|_n-\theta_z|_s}{\Delta z}\right].
        \label{eq:temperature_cell_balance}
\end{equation}
The convective face temperatures are reconstructed with the LUD formula above, while the
diffusive face gradients use central differences. The wall face receives the prescribed heat
flux through the ghost value
$\theta_g=\theta_{i,j}+\Delta r\,\Rey\Pran$; the inlet face uses $\theta=0$ and the
outlet uses a zero axial gradient. Thus every cell is updated from its local radial and axial
neighbour values, together with the local face velocities. In the AB2 time integrator the
resulting cellwise right-hand side is stored and updated as
\begin{equation}
        \theta_{i,j}^{n+1}=\theta_{i,j}^{n}+\Delta t
        \left(\tfrac32 R_{\theta,i,j}^{n}-\tfrac12 R_{\theta,i,j}^{n-1}\right),
\end{equation}
with a forward-Euler startup step when the previous right-hand side is unavailable. Heat
therefore first enters cells adjacent to the wall and then spreads radially and downstream;
the centreline and downstream cells develop later because they receive heat through diffusion
and axial advection rather than directly from the heated wall.

\subsubsection{Axial Momentum (at w-face CV)}

The w-CV at $(r_{c,i},\, z_{f,j})$ spans $[r_{f,i},\, r_{f,i+1}] \times [z_{c,j-1},\, z_{c,j}]$.

\textbf{AB2 predictor step} (including the known pressure gradient):
\begin{equation}
\boxed{
    w^*_{i,j} = w^n_{i,j} + \Delta t\left[
    \frac{3}{2}F^n_{w,i,j}-\frac{1}{2}F^{n-1}_{w,i,j}
    -\frac{p^n_{i,j}-p^n_{i,j-1}}{\Delta z}\right],
    \quad F_w=-C_r^w-C_z^w+D_r^w+D_z^w
}
\label{eq:predictor_w}
\end{equation}

where the convective terms (upwind) are:
\begin{align}
    C_r^w &= \frac{r_{i+\frac{1}{2}}\,u_E\,w_{E,\text{up}} - r_{i-\frac{1}{2}}\,u_I\,w_{I,\text{up}}}{r_{c,i}\,\Delta r} \\
    C_z^w &= \frac{\bar{w}_N\,w_{N,\text{up}} - \bar{w}_S\,w_{S,\text{up}}}{\Delta z}
\end{align}

and the diffusive terms (central) are:
\begin{align}
    D_r^w &= \frac{1}{\Rey}\,\frac{r_{i+\frac{1}{2}}(w_{i+1,j}-w_{i,j}) - r_{i-\frac{1}{2}}(w_{i,j}-w_{i-1,j})}{r_{c,i}\,\Delta r^2} \\
    D_z^w &= \frac{1}{\Rey}\,\frac{w_{i,j+1} - 2w_{i,j} + w_{i,j-1}}{\Delta z^2}
\end{align}

The radial momentum equation is discretized analogously at u-face control volumes.

\subsection{Temporal Discretization}

Explicit second-order Adams--Bashforth time stepping is used. For a spatial
operator $F$, the update is
\begin{align}
    F^{AB2} &= \tfrac{3}{2}F^n-\tfrac{1}{2}F^{n-1}, \\
    \phi^{n+1} &= \phi^n+\Delta t\,F^{AB2}.
\end{align}
The first step uses forward Euler because $F^{n-1}$ is not yet available. The
current momentum and temperature RHS arrays are stored for the following step.
AB2 needs one RHS evaluation and one pressure-Poisson solve per time step.

\paragraph{Why AB2 is second order.} Forward Euler, $\phi^{n+1}=\phi^n+\Delta t\,F^n$, freezes the
right-hand side at its value at $t^n$ and so approximates the average slope over $[t^n,t^{n+1}]$
with $\mathcal{O}(\Delta t)$ error, giving local truncation error $\mathcal{O}(\Delta t^2)$ per
step and global error $\mathcal{O}(\Delta t)$. AB2 instead linearly extrapolates the two most
recent slopes. Expanding $F^{n-1}=F(t^n-\Delta t) = F^n-\Delta t\,\dot F^n+\tfrac12\Delta t^2\ddot
F^n+\mathcal{O}(\Delta t^3)$,
\begin{equation}
    \tfrac32 F^n-\tfrac12 F^{n-1}
    = F^n+\tfrac12\Delta t\,\dot F^n-\tfrac14\Delta t^2\ddot F^n+\mathcal{O}(\Delta t^3),
\end{equation}
which matches the true average slope on $[t^n,t^{n+1}]$, $F^n+\tfrac12\Delta t\,\dot F^n
+\mathcal{O}(\Delta t^2)$, through the $\mathcal{O}(\Delta t)$ term: the leading error in the
slope estimate is pushed from $\mathcal{O}(\Delta t^0)$ (Euler) to $\mathcal{O}(\Delta t^2)$.
Multiplying by $\Delta t$ in the update gives local truncation error $\mathcal{O}(\Delta t^3)$
per step; accumulating over the $N=\mathcal{O}(1/\Delta t)$ steps needed to reach a fixed final
time gives global error $\mathcal{O}(\Delta t^2)$, i.e.\ AB2 is second order. Because the
pressure-Poisson equation is solved directly (Section~\ref{sec:poisson_solver}, no
iterative truncation) at every step, it introduces no additional error beyond machine precision;
combined with the $\mathcal{O}(\Delta x^2)$ spatial operators above, the overall scheme is
therefore second order in \emph{both} space and time, as the assignment requires.

The temperature equation uses the same AB2 extrapolation of its
convection--diffusion operator.

The time step $\Delta t = 0.002$ satisfies the stability constraints:
\begin{itemize}
    \item Convective CFL: $\Delta t / \Delta z = 0.01 \ll 1$
    \item Viscous stability: $\Delta t\,(1/\Rey)\,(1/\Delta r^2 + 1/\Delta z^2) \approx 0.2 < 0.5$
\end{itemize}

\subsection{Pressure Correction Method (Projection)}
\label{sec:pressure-correction}

A segregated approach is used. Each time step consists of:

\begin{enumerate}
    \item \textbf{Predictor}: Extrapolate the convective--diffusive momentum
    RHS with AB2 and include the known pressure gradient:
    \begin{equation}
        \vec{u}^{*}=\vec{u}^{n}+\Delta t\left[
        \tfrac{3}{2}F^n-\tfrac{1}{2}F^{n-1}-\nabla p^n\right].
    \end{equation}

    \item \textbf{Poisson equation}: Find pressure correction $p'$ such that the corrected velocity is divergence-free:
    \begin{equation}
        \frac{1}{r}\parder{}{r}\!\left(r\parder{p'}{r}\right) + \pparder{p'}{z}
        = \frac{1}{\Delta t}\left[\frac{1}{r}\parder{(r\,u^*)}{r} + \parder{w^*}{z}\right]
        \label{eq:poisson}
    \end{equation}

    Discretized:
    \begin{equation}
        \frac{r_{i+\frac{1}{2}}(p'_{i+1,j}-p'_{i,j}) - r_{i-\frac{1}{2}}(p'_{i,j}-p'_{i-1,j})}{r_{c,i}\,\Delta r^2}
        + \frac{p'_{i,j+1} - 2p'_{i,j} + p'_{i,j-1}}{\Delta z^2}
        = b_{i,j}
    \end{equation}

    \item \textbf{Velocity correction}:
    \begin{align}
        u^{n+1}_{i,j} &= u^*_{i,j} - \Delta t\,\frac{p'_{i,j} - p'_{i-1,j}}{\Delta r} \\
        w^{n+1}_{i,j} &= w^*_{i,j} - \Delta t\,\frac{p'_{i,j} - p'_{i,j-1}}{\Delta z}
    \end{align}

    \item \textbf{Pressure update} with under-relaxation:
    \begin{equation}
        p^{n+1} = p^n + \alpha_p\,p', \qquad \alpha_p = 0.7
    \end{equation}
\end{enumerate}

\subsection{Poisson Solver}
\label{sec:poisson_solver}

The assignment requires a \emph{direct} solution method for the pressure Poisson equation, and requires it to be \emph{programmed} rather than delegated to a library routine.
Accordingly no use is made of MATLAB's backslash operator (\texttt{A\textbackslash b}, i.e.\ the UMFPACK sparse LU), of \texttt{decomposition}, or of SciPy's \texttt{spsolve}: the elimination is written out in \texttt{lu\_factor.m}, \texttt{lu\_solve.m}, \texttt{factorize\_poisson.m} and \texttt{solve\_poisson.m}.
Backslash appears in exactly one file in the whole repository, the verification script \texttt{verify\_poisson\_solver.m}, where it serves as an independent reference (Sec.~\ref{sec:poisson_verification}).

In the terminology of the course notes~\cite{Steiner2025}, Sec.~III.3.1, the scheme used here
combines two of the four direct solvers presented there: the Thomas algorithm (TDMA, method~(C)),
in its block form, as the outer elimination, with an LU decomposition (method~(B)) as the kernel
that factorises each block.
The same section notes that the TDMA, although in itself a one-dimensional algorithm, ``is often
used as sub-algorithm of other solving methods for multidimensional, i.e.\ non-tridiagonal,
elliptic systems''~\cite{Steiner2025} --- which is precisely the role it plays here.
The notes present the TDMA for a scalar tridiagonal matrix; the block form used here, in which
the scalar entries are replaced by the $n_r \times n_r$ blocks of Eq.~\eqref{eq:poisson_blocktri},
is the direct extension of that algorithm to the two-dimensional operator.
The iterative alternatives of Sec.~III.3.2 (Gauss--Seidel, SOR, LSOR, ADISOR) are excluded by the
assignment's requirement of a \emph{direct} method.

\paragraph{Block structure.}
Group the $n_r \cdot n_z = 12{,}500$ unknowns by axial station~$j$, so that column~$j$ of the field $p'$ forms one block $\mathbf{p}_j$ of length~$n_r$.
The five-point stencil of Eq.~\eqref{eq:poisson} couples a cell only to its radial neighbours (same~$j$) and to $j \pm 1$, so the operator is \textbf{block tridiagonal} with $n_z = 250$ blocks of size $n_r \times n_r = 50 \times 50$:
\begin{equation}
    \mathbf{S}_j\,\mathbf{p}_{j-1} + \mathbf{D}_j\,\mathbf{p}_j + \mathbf{N}_j\,\mathbf{p}_{j+1} = \mathbf{b}_j,
    \qquad j = 1,\dots,n_z ,
    \label{eq:poisson_blocktri}
\end{equation}
with
\begin{equation}
    \mathbf{D}_j = \operatorname{tridiag}\!\big(w_i,\; -(e_i + w_i + 2a_z),\; e_i\big),
    \qquad
    \mathbf{S}_j = \mathbf{N}_j = a_z\,\mathbf{I} ,
\end{equation}
\begin{equation}
    e_i = \frac{r_{i+\frac{1}{2}}}{r_i\,\Delta r^2},
    \qquad
    w_i = \frac{r_{i-\frac{1}{2}}}{r_i\,\Delta r^2},
    \qquad
    a_z = \frac{1}{\Delta z^2} .
\end{equation}
The boundary conditions enter as modifications of these coefficients rather than as extra equations:
$e_{n_r} = 0$ (Neumann at the wall);
$w_1 = 0$ automatically, because $r_{\frac{1}{2}} = 0$ (Neumann at the centreline);
no south coupling at $j = 1$, so that diagonal carries $a_z$ once instead of twice (Neumann at the inlet);
and $\mathbf{D}_{n_z} = \mathbf{I}$, $\mathbf{S}_{n_z} = \mathbf{N}_{n_z} = \mathbf{0}$ (Dirichlet reference at the outlet).
Figure~\ref{fig:bt_structure} sketches this structure.

\paragraph{Block Thomas elimination.}
Equation~\eqref{eq:poisson_blocktri} is eliminated by the block form of the Thomas algorithm.
The forward elimination forms the Schur complements
\begin{equation}
    \mathbf{M}_1 = \mathbf{D}_1,
    \qquad
    \mathbf{M}_j = \mathbf{D}_j - \mathbf{S}_j\,\mathbf{M}_{j-1}^{-1}\,\mathbf{N}_{j-1}
                 = \mathbf{D}_j - a_z^2\,\mathbf{M}_{j-1}^{-1} ,
    \label{eq:schur}
\end{equation}
after which the solution follows from two sweeps over the axial stations,
\begin{align}
    \mathbf{y}_j &= \mathbf{M}_j^{-1}\big(\mathbf{b}_j - a_z\,\mathbf{y}_{j-1}\big),
        && j = 1,\dots,n_z \quad \text{(forward)}, \\
    \mathbf{p}_j &= \mathbf{y}_j - a_z\,\mathbf{M}_j^{-1}\,\mathbf{p}_{j+1},
        && j = n_z,\dots,1 \quad \text{(back substitution)} .
\end{align}
Each $\mathbf{M}_j$ is factorised with a hand-written Doolittle LU with partial pivoting, $\mathbf{P}\mathbf{M}_j = \mathbf{L}\mathbf{U}$ (\texttt{lu\_factor.m}), and inverted by forward/back substitution against the identity (\texttt{lu\_solve.m}).
Because the unknowns are grouped by axial station, column~$j$ of the $(n_r \times n_z)$ field \emph{is} block~$j$ of the system; the implementation therefore needs no flattening and no flat-index bookkeeping at all.
Figure~\ref{fig:bt_flowchart} shows the resulting algorithm.

\paragraph{Cost.}
The operator is constant in time, so the factorisation~\eqref{eq:schur} is performed once, before the time loop, at a cost of $O(n_z\,n_r^3) \approx 3 \times 10^{7}$ flops --- 0.16\,s measured, in both MATLAB and Python.
This is the advantage of LU-type factorisation stressed in~\cite{Steiner2025}: when the
coefficient matrix does not change --- as here, with a temporally constant mesh --- the costly
factorisation has to be carried out only once, and every subsequent solve needs only the cheap
substitution steps.
Each subsequent Poisson solve is then only the two sweeps above, i.e.\ two $50 \times 50$ matrix--vector products per axial station, $O(n_z\,n_r^2) \approx 1.2 \times 10^{6}$ flops.
Measured: 1.8\,ms per solve in MATLAB and 1.0\,ms in Python, the same order as the backslash solve it replaces (0.5--0.9\,ms) and negligible beside the momentum and energy right-hand-side evaluations.

\paragraph{Stability without pivoting between blocks.}
By construction the diagonal coefficient of Eq.~\eqref{eq:poisson} is minus the sum of the four neighbour coefficients, so $|a_{kk}| = \sum_{l \neq k} |a_{kl}|$ in every interior row, with strict inequality in the Dirichlet outlet rows.
Hence $-\mathbf{A}$ is an irreducibly diagonally dominant M-matrix, a property inherited by every Schur complement $\mathbf{M}_j$ in Eq.~\eqref{eq:schur}.
LU without pivoting is therefore guaranteed to exist and to be backward stable for this operator, which is what makes a hand-written Thomas-type elimination safe here; partial pivoting is retained inside \texttt{lu\_factor.m} regardless, so that the kernel stays generally usable.

Boundary conditions for $p'$:
\begin{itemize}
    \item Centreline ($r=0$): Neumann $\partial p'/\partial r = 0$ (natural: $r_{-\frac{1}{2}} = 0$, flux vanishes)
    \item Wall ($r=0.5$): Neumann $\partial p'/\partial r = 0$
    \item Inlet ($z=0$): Neumann $\partial p'/\partial z = 0$
    \item Outlet ($z=50$): Dirichlet $p' = 0$ is imposed at the last pressure cell centre as the reference. This location is half a cell upstream of the physical outlet face, so the condition is a discrete pressure reference rather than an exactly face-centred outlet condition.
\end{itemize}

\subsection{Key Implementation Details}

\begin{itemize}
    \item \textbf{2nd-order upwind differencing}: Convective fluxes in both momentum and energy equations use LUD. It is formally second-order on smooth, resolved regions, as required by the assignment, but it is unbounded and can produce small local overshoots near sharp transients or boundaries.

    \item \textbf{Hand-written direct solver}: The Poisson operator ($12{,}500$ unknowns) is block tridiagonal --- $250$ blocks of $50 \times 50$ --- and is eliminated by a block Thomas algorithm built on our own Doolittle LU kernel. It is constant in time, so it is factorised once before the time loop; each time step costs only the two sweeps. No library linear solve is used (Sec.~\ref{sec:poisson_solver}).

    \item \textbf{Vectorised operations}: All stencil computations are performed using MATLAB's array operations, avoiding explicit loops over grid cells for efficiency.

    \item \textbf{Staggered grid}: Prevents checkerboard pressure oscillations; velocity components are stored at the faces where mass fluxes are naturally evaluated.

    \item \textbf{Ghost cells}: Velocity and temperature boundary conditions are implemented through ghost-cell extensions. Pressure-correction boundary conditions are imposed directly in the block coefficients of the Poisson operator, including the last-cell-centre outlet reference.
\end{itemize}

\section{Analytical Solution}

For fully developed flow (large $z$), analytical solutions exist for comparison.

\subsection{Velocity Profile (Hagen--Poiseuille)}

\begin{equation}
    \frac{w}{W_{\text{in}}} = 2\left[1 - (2r^*)^2\right]
    \label{eq:analytical_velocity}
\end{equation}

The centreline velocity is $w_{\max} = 2\,W_{\text{in}}$.
The hydrodynamic entry length is approximately $L_h \approx 0.05\,\Rey\,D = 5D$.

\subsection{Temperature Profile}

For constant wall heat flux in fully developed thermal flow:
\begin{equation}
    \theta = 4z^* + \Rey\Pran\left[\frac{1}{2}(2r^*)^2 - \frac{1}{8}(2r^*)^4 - \frac{7}{48}\right]
    \label{eq:analytical_temperature}
\end{equation}
This form follows from direct integration of the non-dimensional energy
equation~\eqref{eq:energy_nd} in the fully developed regime (Section~\ref{sec:nusselt-check}),
subject to the wall condition $\lambda\,\partial T/\partial r|_{\text{wall}}=q_w$ and the global
energy balance $\theta_{\text{bulk}}(z^*) = 4z^*$; see Section~\ref{sec:nusselt-check} for the
full derivation and literature sources.

The thermal entry length is $L_t \approx 0.05\,\Rey\,\Pran\,D = 25D$.

\subsection{Validity of the Fully-Developed Temperature Profile}
\label{sec:fd-validity}

Equation~\eqref{eq:analytical_temperature} makes two logically separate claims, which hold over
different ranges of $z^*$ because they rest on different physics: one about the
cross-sectional \emph{level} of the temperature (the bulk value), and one about its \emph{radial
shape}.

\paragraph{The bulk temperature is exact from the inlet.}
The bulk (mixing-cup) temperature is the flow-weighted cross-sectional average
\begin{equation}
    \theta_{\text{bulk}}(z^*) =
        \frac{\displaystyle\int w^*\,\theta \; \mathrm{d}A}{\displaystyle\int w^* \, \mathrm{d}A}.
\end{equation}
Its axial evolution follows from an energy balance on a thin slice of pipe --- enthalpy gained
by the stream equals heat entering through the wall:
\begin{equation}
    \dot{m}\,c_v\,\parder{T_{\text{bulk}}}{z} = q_w\,\pi D .
\end{equation}
Every factor on the right is constant ($q_w$ is constant and applied from $z=0$; the perimeter
is fixed), and $\dot m$, $c_v$ are constant, so $\partial T_{\text{bulk}}/\partial z$ is constant
along the \emph{entire} pipe. Non-dimensionalized this is $\partial\theta_{\text{bulk}}/\partial
z^* = 4$, and integrating from $\theta_{\text{bulk}}(0)=0$ gives
\begin{equation}
    \theta_{\text{bulk}}(z^*) = 4z^* \qquad\text{exactly, for all } z^*\ge 0 ,
\end{equation}
developed or not. This balance involves only totals (total wall heat input, total enthalpy
flux); it never refers to the shape of the radial profile, so the thermal entrance region does
not perturb it. The one effect that would --- axial conduction near the inlet --- is neglected
here, as specified.

\paragraph{The radial shape requires thermally developed flow.}
Reducing the energy equation~\eqref{eq:energy_nd} to the closed $r^*$-form
in~\eqref{eq:analytical_temperature} assumes the profile is \emph{thermally fully developed}:
self-similar in shape, merely translating downstream, so
$\partial(\theta-\theta_{\text{bulk}})/\partial z^* = 0$. That is exactly the step that makes
$\partial\theta/\partial z^*$ the $r^*$-independent constant $4$ and collapses the PDE to an ODE
in $r^*$. Near the inlet it is false: at $z^*=0$ the temperature is uniform ($\theta=0$, flat);
the developed shape has the wall hotter than the centreline by $\tfrac{11}{48}\Rey\Pran$; and
over the thermal entry length $L_t\approx 25D$ the profile morphs continuously between the two,
a thin heated layer growing inward from the wall while the core stays at inlet temperature.
Through this region $\partial(\theta-\theta_{\text{bulk}})/\partial z^*$ is large and
$r^*$-dependent, and the ODE solution does not apply. Evaluated there anyway,
\eqref{eq:analytical_temperature} gives at the centreline
$4z^* + \Rey\Pran(-\tfrac{7}{48}) \approx 4z^* - 73$, i.e.\ negative for $z^*\lesssim 18$ ---
claiming the core is colder than the inlet. The numerical solution instead holds the core near
$\theta=0$ until the thermal layer reaches it ($z^*\approx 25$), then bends toward the analytical
curve, the two nearly coinciding by $z^*=50$ (Fig.~\ref{fig:T_profiles}). By contrast the
hydrodynamic entry length is only $L_h\approx 0.05\,\Rey\,D \approx 5D$: velocity develops about
five times sooner, since here momentum diffuses $\Pran=5$ times faster than heat.

\paragraph{Why the level can be right while the shape is not.}
The constant $-\tfrac{7}{48}$ is fixed by requiring the flow-weighted average of the radial
bracket to vanish, so \eqref{eq:analytical_temperature} is \emph{built} to satisfy
$\theta_{\text{bulk}}=4z^*$; its mean is pinned to the correct value even where its distribution
about that mean is wrong. This mirrors the Hagen--Poiseuille
profile~\eqref{eq:analytical_velocity}, which always carries the correct flow rate (mean
$w^*=1$ by construction) although the true profile near the inlet is plug-like, not parabolic
--- right integral, wrong shape. The fully-developed profile is therefore used only as the
$z^*\!\to\!\infty$ reference it is: its departure from the numerical solution at small $z^*$ is
expected behaviour, not a defect, and resolving the entrance region is the numerical solution's
role.

\subsection{Nusselt Number}
\label{sec:nusselt-check}

For hydrodynamically and thermally fully developed laminar flow in a circular pipe with
\emph{constant wall heat flux}, the Nusselt number is the exact rational value
\begin{equation}
    \text{Nu} = \frac{h D}{\lambda} = \frac{q_w D}{\lambda\,(T_{\text{wall}} - T_{\text{bulk}})}
              = \frac{\Rey\Pran}{\theta_{\text{wall}} - \theta_{\text{bulk}}}
              = \frac{48}{11} \approx 4.3636 ,
    \label{eq:nusselt}
\end{equation}
a standard textbook result for this thermal boundary condition
\cite{ShahLondon1978,Incropera}.

\paragraph{Derivation.} In the fully developed region $u^*=0$ and the profile translates
rigidly in $z^*$, so the non-dimensional energy equation~\eqref{eq:energy_nd} reduces to
\begin{equation}
    w^*(r^*)\,\parder{\theta}{z^*}
      = \frac{1}{\Rey\Pran}\,\frac{1}{r^*}\parder{}{r^*}\!\left(r^*\parder{\theta}{r^*}\right).
    \label{eq:fd_energy}
\end{equation}
A global energy balance over a pipe slice fixes the axial gradient: with the wall adding heat at
non-dimensional rate $\partial\theta/\partial r^*|_{\text{wall}}=\Rey\Pran$ and the flow carrying
mean velocity $\langle w^*\rangle = 1$, the bulk (mixing-cup) temperature satisfies
$\theta_{\text{bulk}}(z^*) = 4z^*$ exactly, hence $\partial\theta/\partial z^* = 4$ everywhere in
the cross-section. Substituting this and $w^*=2[1-(2r^*)^2]$ into~\eqref{eq:fd_energy} and
integrating twice, with $\partial\theta/\partial r^*|_{r^*=0}=0$ and
$\partial\theta/\partial r^*|_{r^*=1/2}=\Rey\Pran$,
\begin{equation}
    \theta(r^*,z^*) = 4z^* + \Rey\Pran\left[\tfrac12(2r^*)^2 - \tfrac18(2r^*)^4 + c\right] ,
\end{equation}
and the constant $c$ follows from requiring the flow-weighted average of the bracket to vanish
(so that $\theta_{\text{bulk}}=4z^*$), which gives $c=-\tfrac{7}{48}$ --- recovering
Eq.~\eqref{eq:analytical_temperature}. Evaluating the bracket at the wall gives
$\theta_{\text{wall}}-\theta_{\text{bulk}} = \tfrac{11}{48}\,\Rey\Pran$, and~\eqref{eq:nusselt}
follows.

\paragraph{Consistency check.} The value is exact: \eqref{eq:fd_energy} with its boundary
conditions is a linear problem with no free parameters, so $\text{Nu}$ is independent of $\Rey$,
$\Pran$, $q_w$, the fluid properties and the grid. Taking instead the assignment sheet's literal
radial coefficients $(2r^*)^2-\tfrac14(2r^*)^4-\tfrac34$ yields
$\theta_{\text{wall}}-\theta_{\text{bulk}} = \tfrac{11}{24}\,\Rey\Pran$ and hence
$\text{Nu}=24/11\approx2.18$ --- exactly half of $48/11$, and inconsistent both with this
classical result and with the assignment's own stated wall condition (that literal profile has
wall gradient $2\,\Rey\Pran$, and its bulk temperature is not $4z^*$).

\paragraph{Empirical check against the CFD solution.} The solver imposes the physically
consistent wall condition $\partial\theta/\partial r^*=\Rey\Pran$. From the converged field
($t^*=139.728$, $R_\theta<10^{-6}$) the flow-weighted bulk temperature is evaluated at three
downstream stations as
\begin{equation}
    \theta_{\text{bulk}} = \frac{\sum_i w_{i}\,\theta_{i}\,r_{c,i}}{\sum_i w_{i}\,r_{c,i}},
\end{equation}
with $w_i$ interpolated to the cell centres. The wall temperature is taken at the wall face,
consistent with the ghost-cell boundary condition,
$\theta_{\text{wall}} = \theta_{n_r} + \tfrac12\Delta r\,\Rey\Pran$, and
$\text{Nu}=\Rey\Pran/(\theta_{\text{wall}}-\theta_{\text{bulk}})$. The analytical fully developed
values are $\theta_{\text{wall}}-\theta_{\text{bulk}}=\tfrac{11}{48}\Rey\Pran=114.58$ and
$\text{Nu}=48/11=4.364$.

\begin{center}
\begin{tabular}{cccccc}
\toprule
$z^*$ & $\theta_{\text{bulk}}$ (CFD) & $4z^*$ (expected) & $\theta_{\text{wall}}-\theta_{\text{bulk}}$ & $\text{Nu}$ (CFD) & deviation from $48/11$ \\
\midrule
29.9 & 119.59 & 119.6 & 112.22 & 4.455 & $+2.1\%$ \\
39.9 & 159.59 & 159.6 & 113.72 & 4.397 & $+0.8\%$ \\
48.9 & 195.59 & 195.6 & 114.22 & 4.378 & $+0.3\%$ \\
\bottomrule
\end{tabular}
\end{center}

The global energy balance $\theta_{\text{bulk}}=4z^*$ is satisfied to $0.01\%$ at all three
stations, as required for a converged steady field. The CFD Nusselt number decreases
monotonically along the pipe and approaches $48/11$ from above, reaching $+0.3\%$ at
$z^*=48.9$. This is the expected thermal entrance-region behaviour: in the region shortly
after the thermal entry length $L_t\approx25D$ the heated layer near the wall is still thinner
than in the fully developed state, so the wall-to-bulk temperature difference is smaller and
$\text{Nu}$ is higher. (Evaluating $\theta_{\text{wall}}$ at the last cell centre instead of the
wall face would bias $\theta_{\text{wall}}-\theta_{\text{bulk}}$ low by
$\tfrac12\Delta r\,\Rey\Pran=2.5$ and give $\text{Nu}=4.48$ at $z^*=48.9$. An earlier,
unconverged snapshot at $t^*=60$ gave $\theta_{\text{bulk}}$ up to $7\%$ below $4z^*$ near the
outlet, which shows that the full thermal convergence, beyond the radial diffusion timescale
$\sim\Rey\Pran R^{*2}\approx125$, is needed for this comparison.) This is a model-free
confirmation, built only from the governing PDE and the wall BC, that the physical answer is
$48/11$ and not the $2.18$ implied by the assignment-literal coefficients.

\paragraph{Why the solver can never reproduce $24/11$.} The energy equation, the wall BC, and the
inlet condition $\theta_{\text{in}}=0$ are all \emph{linear} in $\theta$, and $\text{Nu}$ in
\eqref{eq:nusselt} is normalised by the wall flux. Multiplying the imposed wall flux by any
constant $k$ simply scales the entire $\theta$ field by $k$, leaving $\text{Nu}$ unchanged at
$48/11$. A CFD run therefore cannot reproduce $24/11$: that number exists only as a property of
the internally inconsistent literal formula, never of a correctly posed simulation
\cite{ShahLondon1978,Incropera,KaysCrawford,BirdStewartLightfoot}.

\subsection{Convergence History}

The simulation targets convergence when the continuity, velocity, and temperature residuals fall
below $10^{-6}$:
\begin{equation}
    R_{\text{cont}} = \max_{i,j}|\nabla\cdot\mathbf{u}|, \quad
    R_{\text{vel}} = \frac{1}{\Delta t}\max\!\left(
        \max|u^{n+1}-u^n|,\max|w^{n+1}-w^n|\right), \quad
    R_{\theta} = \frac{\max|\theta^{n+1}-\theta^n|}{\Delta t}
\end{equation}

\begin{figure}[H]
    \centering
    \includegraphics[width=0.8\textwidth]{../AdamsBashfort2/Plots_matlab/convergence.png}
    \caption{Convergence history: residuals versus time step.}
    \label{fig:convergence}
\end{figure}

The continuity residual (blue) is flat from the very first step rather than decaying over time:
the incremental pressure correction re-solves the Poisson equation to machine precision every
single step, so $\nabla\cdot\mathbf{u}$ is driven to its numerical floor ($\sim10^{-14}$--$10^{-15}$)
at each iteration by construction, independent of whether the flow itself has reached steady
state. $R_{\text{vel}}$ (red), in contrast, measures how much the velocity field is still
changing between steps: it decays exponentially, by roughly one decade every $1000$ steps
($\Delta t^*=2$), until it reaches the round-off floor at step $\approx15{,}000$
($t^*\approx30$) -- only then does it flatten into a noisy plateau rather than continuing
smoothly to zero. That plateau is round-off noise: $R_{\text{vel}}$ divides an already
near-machine-precision difference
$u^{n+1}-u^n$ by $\Delta t=0.002$, amplifying floating-point noise by a factor $1/\Delta t=500$,
which is why it appears jagged at $\mathcal{O}(10^{-13})$ while $R_{\text{cont}}$ (not divided by
$\Delta t$) stays visually flat near $\mathcal{O}(10^{-14})$--$\mathcal{O}(10^{-15})$.

In the extended MATLAB run, $R_{\text{cont}}=9.90\times10^{-15}$ and
$R_{\text{vel}}=2.78\times10^{-14}$ at the converged time $t^*=139.728$. The temperature residual
decreases from approximately $101$ at the first step and reaches
$R_\theta=9.99\times10^{-7}$ at step $69{,}864$, just below the target $10^{-6}$. This residual is
the maximum local rate of change, not the error relative to the analytical temperature profile. Its
earlier non-monotone decrease was therefore not a numerical failure: as the heated layer moved
downstream and diffused inward, the cell containing the largest change moved as well. The dominant
timescale is thermal diffusion across the radius, $\Rey\Pran R^{*2}\approx125$, which explains why
the temperature required substantially longer to settle than the velocity. The final result is
therefore a converged finite-volume solution at the stated residual tolerance, while the analytical
fully-developed temperature profile remains a separate asymptotic reference for radial shape.

\section{Conclusion}

A finite volume solver for 2D axisymmetric incompressible pipe flow was developed in MATLAB and validated against analytical solutions.
The solver uses a staggered grid with 2nd-order upwind (linear upwind) differencing for convective stability, explicit second-order Adams--Bashforth time integration, and an incremental projection method with one sparse direct Poisson solve per time step.

The numerical results reproduce the expected physics:
\begin{itemize}
    \item The velocity field develops from a uniform inlet profile to the parabolic Hagen--Poiseuille profile within the hydrodynamic entry length. At $z^*=50$ the numerical and analytical
          Hagen--Poiseuille profiles agree to within $\mathcal{O}(10^{-4})$ (Fig.~\ref{fig:w_comparison}) -- the velocity field is essentially fully converged and validated.
    \item The temperature field shows the characteristic linear rise of the bulk temperature along the pipe axis ($\theta_{\text{bulk}}=4z^*$, confirmed to within $0.01\%$ at $z^*=29.9$, $39.9$ and $48.9$; see
          Section~\ref{sec:nusselt-check}) and a radial profile that approaches the analytical fully-developed solution as $z^*$ increases (Fig.~\ref{fig:T_profiles}), as expected since that solution
          is only valid asymptotically. The local Nusselt number decreases along the pipe towards $48/11$ and is within $0.3\%$ of it at $z^*=48.9$; at smaller $z^*$ the difference is primarily the thermal entrance-region shape, while the final field satisfies the temporal convergence criterion.
\end{itemize}

\section{Temperature Formula and Nusselt Number Audit}
\label{app:nusselt}

This appendix derives from first principles why the fully developed Nusselt number for the
present problem is exactly $\text{Nu}=48/11\approx4.3636$, and shows which Nusselt number each of
the two candidate fully developed temperature profiles implies: the one printed on the assignment
sheet and the corrected one used throughout this report and the code.

All quantities are non-dimensional in the scaling of the assignment:
\begin{equation}
    r^*=\frac{r}{D},\quad
    z^*=\frac{z}{D},\quad
    w^*=\frac{w}{W_{\text{in}}},\quad
    \theta=\frac{(T-T_{\text{in}})\rho c_v W_{\text{in}}}{q_w},\quad
    \Rey=\frac{\rho W_{\text{in}}D}{\mu}=100,\quad
    \Pran=\frac{\mu c_v}{\lambda}=5 .
\end{equation}
It is convenient to introduce
\begin{equation}
    s \equiv 2r^* = \frac{2r}{D}\in[0,1],
\end{equation}
so that the centreline is at $s=0$ and the pipe wall at $s=1$.

\subsection{The Two Candidate Fully Developed Profiles}

Both candidates have the form
\begin{equation}
    \theta(r^*,z^*) = 4z^* + \Rey\Pran\,B(s),
\end{equation}
and differ only in the radial bracket $B(s)$:
\begin{align}
    B_{\text{corr}}(s) &= \tfrac12 s^2 - \tfrac18 s^4 - \tfrac{7}{48}
        &&\text{(corrected, used in report and code)},\\
    B_{\text{lit}}(s)  &= s^2 - \tfrac14 s^4 - \tfrac34
        &&\text{(assignment-literal)}.
\end{align}
The corrected bracket is implemented in \texttt{analytical\_temperature.m}
(Listing~\ref{lst:anatemp}) and is Eq.~\eqref{eq:analytical_temperature} of the report.

\subsection{Governing Equation and Axial Gradient}

For hydrodynamically and thermally fully developed axisymmetric flow ($u^*=0$, the \emph{shape}
of the profile no longer changes with $z^*$), with viscous dissipation and axial conduction
neglected, the non-dimensional energy equation reduces to
\begin{equation}
    w^*(r^*)\,\parder{\theta}{z^*}
      =
      \frac{1}{\Rey\Pran}\,
      \frac{1}{r^*}\parder{}{r^*}
      \left(r^*\parder{\theta}{r^*}\right).
    \label{eq:app_fd_energy}
\end{equation}
A global energy balance over a pipe slice fixes the axial gradient. The wall adds heat at the
non-dimensional rate $\partial\theta/\partial r^*|_{\text{wall}}=\Rey\Pran$ per unit wall area
(Eq.~\eqref{eq:app_wall_gradient}), and the enthalpy flux carried by the flow is
$\int w^*\theta\,\mathrm{d}A$. With $\langle w^*\rangle=1$ (the parabolic profile carries the same
mean velocity as the uniform inlet), the bulk (mixing-cup) temperature
\begin{equation}
    \theta_{\text{bulk}} \equiv
    \frac{\int w^*\theta\,\mathrm{d}A}{\int w^*\,\mathrm{d}A}
\end{equation}
satisfies
\begin{equation}
    \frac{\mathrm{d}\theta_{\text{bulk}}}{\mathrm{d}z^*} = 4
    \qquad\Longrightarrow\qquad
    \theta_{\text{bulk}}(z^*) = 4z^*
    \label{eq:app_bulk}
\end{equation}
exactly, at every $z^*$. This is energy conservation from the inlet and is independent of the
radial profile shape. In the fully developed region the whole profile translates uniformly in
$z^*$, so
\begin{equation}
    \parder{\theta}{z^*} = \frac{\mathrm{d}\theta_{\text{bulk}}}{\mathrm{d}z^*} = 4
\end{equation}
everywhere in the cross-section.

\subsection{Velocity Profile and Wall Boundary Condition}

The Hagen--Poiseuille profile (Eq.~\eqref{eq:analytical_velocity}) is
\begin{equation}
    w^*(r^*) = 2\left[1-(2r^*)^2\right] = 2\,(1-s^2).
    \label{eq:app_velocity}
\end{equation}
Fourier's law at the heated wall,
\begin{equation}
    \lambda \parder{T}{r}\bigg|_{\text{wall}} = q_w ,
\end{equation}
non-dimensionalises directly:
\begin{equation}
    \parder{\theta}{r^*}\bigg|_{\text{wall}}
    =
    D\,\frac{\rho c_v W_{\text{in}}}{q_w}\,
    \parder{T}{r}\bigg|_{\text{wall}}
    =
    D\,\frac{\rho c_v W_{\text{in}}}{q_w}\cdot\frac{q_w}{\lambda}
    =
    \frac{\rho c_v W_{\text{in}}D}{\lambda}
    =
    \Rey\Pran .
    \label{eq:app_wall_gradient}
\end{equation}
Numerically $\Rey\Pran=500$ here, not an order-one value: the reference length and temperature
scales do not cancel. The symmetry condition at the axis is
\begin{equation}
    \parder{\theta}{r^*}\bigg|_{r^*=0} = 0 .
\end{equation}

\subsection{Integration of the Radial Equation}

Inserting Eqs.~\eqref{eq:app_bulk} and~\eqref{eq:app_velocity} into
Eq.~\eqref{eq:app_fd_energy} gives
\begin{equation}
    \frac{1}{r^*}\frac{\mathrm{d}}{\mathrm{d}r^*}
    \left(r^*\frac{\mathrm{d}\theta}{\mathrm{d}r^*}\right)
    =
    \Rey\Pran\cdot4\cdot w^*(r^*)
    =
    8\Rey\Pran\left(1-4r^{*2}\right).
\end{equation}
Integrating once and applying regularity ($r^*\,\mathrm{d}\theta/\mathrm{d}r^*\to0$ as
$r^*\to0$),
\begin{align}
    r^*\frac{\mathrm{d}\theta}{\mathrm{d}r^*}
    &= 8\Rey\Pran\left(\frac{r^{*2}}{2}-r^{*4}\right),\\
    \frac{\mathrm{d}\theta}{\mathrm{d}r^*}
    &= 8\Rey\Pran\,r^*\left(\frac12-r^{*2}\right).
    \label{eq:app_first_integral}
\end{align}
At the wall, $r^*=\tfrac12$,
\begin{equation}
    \frac{\mathrm{d}\theta}{\mathrm{d}r^*}\bigg|_{r^*=1/2}
    = 8\Rey\Pran\cdot\frac12\cdot\left(\frac12-\frac14\right)
    = \Rey\Pran ,
\end{equation}
which matches the wall condition~\eqref{eq:app_wall_gradient}. Integrating a second time,
\begin{equation}
    \theta(r^*,z^*)
    = 2\Rey\Pran\left(r^{*2}-r^{*4}\right) + C(z^*)
    = \Rey\Pran\left(\frac12 s^2-\frac18 s^4\right) + C(z^*),
    \qquad s=2r^* .
    \label{eq:app_second_integral}
\end{equation}

\subsection{Integration Constant from the Global Energy Balance}

Write $C(z^*)=4z^*+\Rey\Pran\,c$ with $c$ a pure constant. The energy balance
$\theta_{\text{bulk}}=4z^*$ (Eq.~\eqref{eq:app_bulk}) requires the flow-weighted mean of the
radial bracket to vanish:
\begin{equation}
    \left\langle \frac12 s^2-\frac18 s^4 + c \right\rangle_w = 0,
    \qquad
    \langle f\rangle_w \equiv
    \frac{\int_0^{1/2} f\,w^*r^*\,\mathrm{d}r^*}{\int_0^{1/2} w^*r^*\,\mathrm{d}r^*}.
\end{equation}
With $r^*=s/2$ and Eq.~\eqref{eq:app_velocity},
\begin{equation}
    w^*r^*\,\mathrm{d}r^* = \frac12\,s\,(1-s^2)\,\mathrm{d}s,
    \qquad
    \int_0^1 \frac12\,s\,(1-s^2)\,\mathrm{d}s = \frac18 ,
\end{equation}
so that
\begin{equation}
    \left\langle \frac12 s^2-\frac18 s^4 \right\rangle_w
    = 8\int_0^1\left(\frac12 s^2-\frac18 s^4\right)\frac12\,s\,(1-s^2)\,\mathrm{d}s
    = 8\cdot\frac{7}{384}
    = \frac{7}{48}.
\end{equation}
Hence $c=-7/48$, and the corrected fully developed profile is
\begin{equation}
    \theta(r^*,z^*) =
    4z^* + \Rey\Pran\left[\frac{1}{2}(2r^*)^2
    - \frac{1}{8}(2r^*)^4 - \frac{7}{48}\right].
    \label{eq:app_corrected_temperature}
\end{equation}
Naively halving the assignment's constant $-\tfrac34$ would give $-\tfrac38$, which is wrong: the
constant must be re-derived from the energy balance and equals $-\tfrac{7}{48}$.

\subsection{Nusselt Number of the Corrected Profile}

The Nusselt number is $\text{Nu}=hD/\lambda$ with
$h=q_w/(T_{\text{wall}}-T_{\text{bulk}})$. Non-dimensionalising the wall-to-bulk temperature
difference,
\begin{equation}
    T_{\text{wall}}-T_{\text{bulk}}
    = \left(\theta_{\text{wall}}-\theta_{\text{bulk}}\right)
      \frac{q_w}{\rho c_v W_{\text{in}}},
\end{equation}
gives
\begin{equation}
    \text{Nu} =
    \frac{q_w D}{\lambda\left(T_{\text{wall}}-T_{\text{bulk}}\right)}
    =
    \frac{\rho c_v W_{\text{in}}D/\lambda}{\theta_{\text{wall}}-\theta_{\text{bulk}}}
    =
    \frac{\Rey\Pran}{\theta_{\text{wall}}-\theta_{\text{bulk}}}.
    \label{eq:app_nusselt_def}
\end{equation}
From Eq.~\eqref{eq:app_corrected_temperature}, the bracket at the wall ($s=1$) is
\begin{equation}
    B_{\text{corr}}(1) = \frac12-\frac18-\frac{7}{48}
    = \frac{24}{48}-\frac{6}{48}-\frac{7}{48}
    = \frac{11}{48},
\end{equation}
while $\theta_{\text{bulk}}$ corresponds to a bracket value of $0$. Therefore
\begin{equation}
    \theta_{\text{wall}}-\theta_{\text{bulk}}
    = \frac{11}{48}\Rey\Pran,
    \qquad
    \text{Nu} = \frac{\Rey\Pran}{(11/48)\Rey\Pran} = \frac{48}{11} \approx 4.3636 .
    \label{eq:app_nusselt_corr}
\end{equation}

\subsection{Why the Value is Exactly 48/11}

Equations~\eqref{eq:app_fd_energy}--\eqref{eq:app_wall_gradient} form a linear two-point
boundary-value problem with fixed coefficients and no free parameters. Given only
\begin{itemize}
    \item hydrodynamically \emph{and} thermally fully developed flow,
    \item constant wall heat flux (the H1 thermal boundary condition),
    \item laminar flow with the parabolic velocity profile~\eqref{eq:app_velocity},
    \item constant fluid properties,
    \item negligible axial conduction and viscous dissipation,
\end{itemize}
the radial profile~\eqref{eq:app_corrected_temperature} and the ratio~\eqref{eq:app_nusselt_corr}
are completely determined. $\Rey$, $\Pran$, $q_w$, $D$, the fluid properties and the numerical
grid all cancel out of $\text{Nu}$: they rescale $\theta$ but not the dimensionless ratio
$(\theta_{\text{wall}}-\theta_{\text{bulk}})/(\Rey\Pran)$. The value $48/11$ is a rational number
obtained by integration, not an empirical correlation, and it is the classical result for this
configuration \cite{ShahLondon1978,Incropera,KaysCrawford,BirdStewartLightfoot}. For contrast, a
constant wall \emph{temperature} gives the different value $\text{Nu}\approx3.66$ (from the first
eigenvalue of the Graetz problem); $48/11$ is specifically the constant-heat-flux value relevant
to this assignment.

\subsection{Assignment-Literal Profile: Nu = 24/11}

Taking the printed bracket $B_{\text{lit}}(s)=s^2-\tfrac14s^4-\tfrac34$ at face value:

\paragraph{Wall value.}
\begin{equation}
    B_{\text{lit}}(1) = 1-\frac14-\frac34 = 0
    \qquad\Longrightarrow\qquad
    \theta_{\text{wall}} = 4z^* .
\end{equation}

\paragraph{Bulk value.} With the same flow-weighted average,
\begin{align}
    \left\langle s^2-\frac14 s^4 \right\rangle_w
    &= 8\int_0^1\left(s^2-\frac14 s^4\right)\frac12\,s\,(1-s^2)\,\mathrm{d}s
     = 8\cdot\frac{7}{192} = \frac{7}{24},\\
    \left\langle B_{\text{lit}} \right\rangle_w
    &= \frac{7}{24}-\frac34 = \frac{7}{24}-\frac{18}{24} = -\frac{11}{24}.
\end{align}
The literal formula therefore gives
\begin{equation}
    \theta_{\text{bulk,lit}} = 4z^* - \frac{11}{24}\Rey\Pran \neq 4z^*,
\end{equation}
i.e.\ it violates the global energy balance~\eqref{eq:app_bulk}, a second, independent symptom of
the same defect.

\paragraph{Nusselt number.} From Eq.~\eqref{eq:app_nusselt_def},
\begin{align}
    \theta_{\text{wall}}-\theta_{\text{bulk}}
    &= 0-\left(-\frac{11}{24}\Rey\Pran\right) = \frac{11}{24}\Rey\Pran,\\
    \text{Nu}_{\text{lit}} &= \frac{\Rey\Pran}{(11/24)\Rey\Pran}
    = \frac{24}{11} \approx 2.1818 = \frac12\cdot\frac{48}{11}.
\end{align}

\paragraph{Origin of the factor of two.} The literal radial coefficients are $(1,\tfrac14)$ where
the integration~\eqref{eq:app_second_integral} gives $(\tfrac12,\tfrac18)$, each doubled.
Equivalently, the wall slope of the literal profile is
\begin{equation}
    \frac{\mathrm{d}}{\mathrm{d}r^*}\left[\Rey\Pran\left(s^2-\frac14 s^4\right)\right]_{s=1}
    = \Rey\Pran\left(4s-2s^3\right)\Big|_{s=1}
    = 2\Rey\Pran ,
\end{equation}
twice the physical wall gradient $\Rey\Pran$ required by the assignment's own stated condition
$\lambda\,\partial T/\partial r|_{\text{wall}}=q_w$ (Eq.~\eqref{eq:app_wall_gradient}). A wall
gradient twice as steep produces a wall-to-bulk temperature difference twice as large and, since
$\text{Nu}\propto1/(\theta_{\text{wall}}-\theta_{\text{bulk}})$, a Nusselt number half as large.

Thus the assignment-literal coefficients are not an alternative physical solution; they are a
transcription with the radial terms doubled, equivalently missing the factor $\tfrac12$ on
$\Rey\Pran$, and with $-\tfrac34$ in place of the energy-balance constant $-\tfrac{7}{48}$.
Because the temperature equation, the wall boundary condition and the inlet condition
$\theta_{\text{in}}=0$ are linear in $\theta$, multiplying the imposed wall heat flux by any
constant $k'$ only scales the whole temperature field, $\theta\to k'\theta$. The flux-normalised
Nusselt number~\eqref{eq:app_nusselt_def} therefore remains $48/11$ for the correctly posed
problem, and no simulation can reproduce $24/11$.
